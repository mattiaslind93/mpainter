cmake_minimum_required(VERSION 3.31)
project(mpainter_windows_uuid_repro LANGUAGES CXX)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core)
add_executable(uuid-ambiguous EXCLUDE_FROM_ALL main.cc)
add_executable(uuid-explicit EXCLUDE_FROM_ALL main.cc)
target_compile_definitions(uuid-explicit PRIVATE EXPLICIT_UUID_RESET=1)
foreach(target uuid-ambiguous uuid-explicit)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_link_libraries(${target} PRIVATE Qt6::Core)
endforeach()
