cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(bts_spsm_semantic LANGUAGES C CXX)
  include(CTest)
  add_subdirectory(.. reader)
endif()
add_library(bts-spsm-semantic STATIC SpsmSemantic.cc)
target_compile_features(bts-spsm-semantic PUBLIC cxx_std_20)
target_include_directories(bts-spsm-semantic PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(bts-spsm-semantic PUBLIC bts-spsm-reader)
add_executable(bts-spsm-inspect SpsmInspect.cc)
target_link_libraries(bts-spsm-inspect PRIVATE bts-spsm-semantic)
if(BUILD_TESTING)
  add_executable(bts-spsm-semantic-tests ../../tests/spsm_semantic_test.cc)
  target_link_libraries(bts-spsm-semantic-tests PRIVATE bts-spsm-semantic)
  add_test(NAME spsm-semantic-boundaries COMMAND bts-spsm-semantic-tests)
  if(BTS_SPSM_FIXTURE_DIR)
    add_test(NAME spsm-semantic-five-user-files COMMAND bts-spsm-semantic-tests ${BTS_SPSM_FIXTURE_DIR})
  endif()
endif()
