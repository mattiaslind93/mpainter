cmake_minimum_required(VERSION 3.20)
project(native_export_guard LANGUAGES CXX)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui)
find_package(ZLIB REQUIRED)
set(BTS_EXPORT_CONFIG "" CACHE FILEPATH "Private frozen libbts-export-config.a from a completed host build")
if(NOT EXISTS "${BTS_EXPORT_CONFIG}")
 message(FATAL_ERROR "Set BTS_EXPORT_CONFIG to a frozen model archive")
endif()
add_executable(native-export-guard ../native_export_guard_test.cc)
target_include_directories(native-export-guard PRIVATE ../../host)
target_compile_features(native-export-guard PRIVATE cxx_std_20)
target_link_libraries(native-export-guard PRIVATE Qt6::Core Qt6::Gui ${BTS_EXPORT_CONFIG} ZLIB::ZLIB)
enable_testing()
add_test(NAME native-export-guard COMMAND native-export-guard)

add_executable(resource-stack-measurement ../resource_stack_measurement_test.cc)
target_include_directories(resource-stack-measurement PRIVATE ../../host)
target_compile_features(resource-stack-measurement PRIVATE cxx_std_20)
add_test(NAME resource-stack-measurement COMMAND resource-stack-measurement)
