cmake_minimum_required(VERSION 3.20)
project(bts_blend_library_tests LANGUAGES CXX)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
set(CMAKE_CXX_STANDARD 20)
option(BTS_SANITIZE "Address/undefined sanitizer for focused library tests" OFF)
if(BTS_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
set(HOST ${CMAKE_CURRENT_LIST_DIR}/../../host)
add_executable(blend-library-index ../blend_library_index_test.cc)
target_include_directories(blend-library-index PRIVATE ${HOST})
target_link_libraries(blend-library-index PRIVATE Qt6::Core Qt6::Gui)
add_executable(blend-library-browser ../blend_library_browser_test.cc ${HOST}/BlendAssetBrowser.cc ${HOST}/FileDialogs.cc)
target_include_directories(blend-library-browser PRIVATE ${HOST})
target_link_libraries(blend-library-browser PRIVATE Qt6::Widgets Qt6::GuiPrivate)
enable_testing()
add_test(NAME index COMMAND blend-library-index)
add_test(NAME browser COMMAND blend-library-browser)
set_tests_properties(index browser PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;BTS_TEST_OUTPUT=${CMAKE_CURRENT_BINARY_DIR}")
