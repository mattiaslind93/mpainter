cmake_minimum_required(VERSION 3.20)
project(bts_scene_formats LANGUAGES CXX)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Gui)
find_package(assimp CONFIG REQUIRED)
include(../../host/SourceIncludeDirs.cmake)
foreach(kind IN ITEMS bridge formats)
  add_executable(bts-scene-${kind}-tests ../scene_${kind}_test.cc ../../host/SceneBridge.cc)
  target_include_directories(bts-scene-${kind}-tests PRIVATE ${BTS_SOURCE_INCLUDE_DIRS})
  target_compile_features(bts-scene-${kind}-tests PRIVATE cxx_std_20)
  target_compile_options(bts-scene-${kind}-tests PRIVATE "$<IF:$<CXX_COMPILER_ID:MSVC>,/UNDEBUG,-UNDEBUG>")
  target_link_libraries(bts-scene-${kind}-tests PRIVATE Qt6::Gui assimp::assimp)
  add_test(NAME scene-${kind} COMMAND bts-scene-${kind}-tests)
endforeach()
