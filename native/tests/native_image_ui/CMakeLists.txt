cmake_minimum_required(VERSION 3.24)
project(BtsNativeImageUiTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
include(../../host/SourceIncludeDirs.cmake)
enable_testing()
option(BTS_TEST_SANITIZERS "Enable address/undefined sanitizers" OFF)
if(BTS_TEST_SANITIZERS)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_executable(document-edit-commit ../document_edit_commit_test.cc)
target_include_directories(document-edit-commit PRIVATE ${BTS_SOURCE_INCLUDE_DIRS})
add_test(NAME document-edit-commit COMMAND document-edit-commit)
add_executable(native-image-panel ../native_material_image_panel_test.cc ../../host/materials/native/NativeMaterialImagePanel.cc)
target_include_directories(native-image-panel PRIVATE ${BTS_SOURCE_INCLUDE_DIRS})
target_link_libraries(native-image-panel PRIVATE Qt6::Widgets)
add_test(NAME native-image-panel COMMAND native-image-panel)
set_tests_properties(native-image-panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
