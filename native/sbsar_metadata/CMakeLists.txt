cmake_minimum_required(VERSION 3.20)
project(bts_sbsar_description_metadata LANGUAGES CXX)
include(CTest)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core)
add_library(bts-sbsar-description STATIC Description.cc)
target_compile_features(bts-sbsar-description PUBLIC cxx_std_20)
target_include_directories(bts-sbsar-description PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(bts-sbsar-description PUBLIC Qt6::Core)
add_executable(bts-sbsar-metadata-dump dump.cc)
target_link_libraries(bts-sbsar-metadata-dump PRIVATE bts-sbsar-description)
option(BTS_METADATA_SANITIZE "Address/undefined sanitizers for the isolated XML module" OFF)
if(BTS_METADATA_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  # Explicitly instrument targets already declared above as well.
  foreach(t bts-sbsar-description bts-sbsar-metadata-dump)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endforeach()
  add_link_options(-fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
  add_executable(bts-sbsar-metadata-tests ../tests/sbsar_metadata_test.cc)
  target_link_libraries(bts-sbsar-metadata-tests PRIVATE bts-sbsar-description)
  add_test(NAME description-xml-boundaries COMMAND bts-sbsar-metadata-tests)
  add_test(NAME description-cli-requires-xml COMMAND bts-sbsar-metadata-dump --sbsar not-an-input)
  set_tests_properties(description-cli-requires-xml PROPERTIES WILL_FAIL TRUE)
  set(BTS_SBSAR_XML_FIXTURE "" CACHE FILEPATH "Optional privately extracted description XML (never graph payload)")
  set(BTS_SBSAR_ARCHIVE_FIXTURE "" CACHE FILEPATH "Optional original archive for opaque before/after hash only")
  if(BTS_SBSAR_XML_FIXTURE)
    add_test(NAME description-actual-xml COMMAND bts-sbsar-metadata-tests --fixture
      "${BTS_SBSAR_XML_FIXTURE}" "${BTS_SBSAR_ARCHIVE_FIXTURE}")
    add_test(NAME description-actual-xml-dump COMMAND bts-sbsar-metadata-dump --xml "${BTS_SBSAR_XML_FIXTURE}")
  endif()
endif()
